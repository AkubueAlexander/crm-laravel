<?php

namespace App\Console\Commands;

use App\Models\Activity;
use App\Models\User;
use App\Notifications\ActivityOverdueNotification;
use Illuminate\Console\Command;
use Illuminate\Support\Collection;
use Illuminate\Support\Facades\DB;
use Throwable;

class FlagOverdueActivities extends Command
{

    public const NOTIFY_WINDOW_DAYS = 7;

    protected $signature = 'activities:flag-overdue';

    protected $description = 'Flag open activities whose due date has passed (all tenants) and notify owners. Idempotent.';

    public function handle(): int
    {
        $now = now();
        $flagged = 0;
        $notified = 0;


        Activity::withoutTenantScope()
            ->whereNull('completed_at')
            ->whereNull('overdue_at')
            ->whereNotNull('due_at')
            ->where('due_at', '<=', $now)
            ->chunkById(500, function ($rows) use ($now, &$flagged, &$notified) {

                $claimed = DB::transaction(function () use ($rows, $now) {
                    $locked = Activity::withoutTenantScope()
                        ->whereIn('id', $rows->modelKeys())
                        ->whereNull('completed_at')
                        ->whereNull('overdue_at')
                        ->lockForUpdate()
                        ->get();

                    if ($locked->isNotEmpty()) {
                        Activity::withoutTenantScope()
                            ->whereIn('id', $locked->modelKeys())
                            ->update(['overdue_at' => $now]);
                    }

                    return $locked;
                });

                $flagged += $claimed->count();
                $notified += $this->notifyOwners($claimed);
            });

        $this->info("Flagged {$flagged} overdue " . ($flagged === 1 ? 'activity' : 'activities') . '.');

        if ($this->getOutput()->isVerbose()) {
            $this->line("Notified {$notified} owner(s).");
        }

        return self::SUCCESS;
    }

    /** @param Collection<int, Activity> $activities */
    private function notifyOwners(Collection $activities): int
    {
        $ownerIds = $activities->pluck('owner_id')->filter()->unique()->values();

        if ($ownerIds->isEmpty()) {
            return 0;
        }

        $owners = User::query()->whereIn('id', $ownerIds)->get()->keyBy('id');
        $cutoff = now()->subDays(self::NOTIFY_WINDOW_DAYS);
        $sent = 0;

        foreach ($activities as $activity) {
            if ($activity->owner_id === null || $activity->due_at === null || $activity->due_at->lt($cutoff)) {
                continue;
            }

            $owner = $owners->get($activity->owner_id);

            // Never notify across tenants, even if a row's owner_id points elsewhere.
            if ($owner === null || (int) $owner->tenant_id !== (int) $activity->tenant_id) {
                continue;
            }

            try {
                $owner->notify(new ActivityOverdueNotification($activity));
                $sent++;
            } catch (Throwable $e) {
                // A broadcast outage must not stop flagging or other owners' notifications.
                report($e);
            }
        }

        return $sent;
    }
}

<?php

namespace App\Domain\Activities\Actions;

use App\Models\Activity;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Gate;

final class CompleteActivityAction
{
    /**
     * Mark an activity done, or reopen it ($completed = false). Idempotent: completing
     * an already-completed activity keeps its original completed_at and is not an error.
     */
    public function execute(User $actor, Activity $activity, bool $completed = true): Activity
    {
        Gate::forUser($actor)->authorize('activities.complete');

        return DB::transaction(function () use ($activity, $completed) {
            // Re-read under a row lock so two racing requests cannot disagree.
            $row = Activity::query()->lockForUpdate()->findOrFail($activity->getKey());

            $row->forceFill([
                'completed_at' => $completed ? ($row->completed_at ?? now()) : null,

            ])->save();

            return $row;
        });
    }
}

<?php

namespace App\Domain\Products\Actions;

use App\Models\Product;
use App\Models\ProductVariant;
use App\Models\User;
use Illuminate\Database\UniqueConstraintViolationException;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Gate;
use Illuminate\Validation\ValidationException;
use App\Domain\Tenant\TenantContext;


final class SaveProductAction
{

    /**
     * @param array{sku: string, name: string, description?: string|null, unit_price: int|float|string, tax_category?: string, is_active?: bool, variants?: list<array<string, mixed>>} $data
     */


    public function execute(User $actor, ?int $productId, array $data): Product
    {
        Gate::forUser($actor)->authorize('products.manage');

        $tenantId = TenantContext::id();


        $variants = array_values($data['variants'] ?? []);



        $this->assertSkusFree($tenantId, $productId, trim((string) $data['sku']), $variants);

        try {
            return DB::transaction(function () use ($productId, $data, $variants): Product {
                $product = $productId === null
                    ? new Product()
                    : Product::query()->lockForUpdate()->findOrFail($productId);

                $description = trim((string) ($data['description'] ?? ''));

                $product->forceFill([
                    'sku' => trim((string) $data['sku']),
                    'name' => trim((string) $data['name']),
                    'description' => $description === '' ? null : $description,
                    'unit_price' => $this->money($data['unit_price']),
                    'tax_category' => $data['tax_category'] ?? 'standard',
                    'is_active' => (bool) ($data['is_active'] ?? true),
                ])->save();

                $this->syncVariants($product, $variants);

                return $product->load('variants');
            });
        } catch (UniqueConstraintViolationException) {
            // Race guard: the friendly pre-check above lost to a concurrent writer.
            throw ValidationException::withMessages(['sku' => ['This SKU is already in use.']]);
        }
    }

    /** @param list<array<string, mixed>> $variants */
    private function syncVariants(Product $product, array $variants): void
    {
        $existing = ProductVariant::query()
            ->where('product_id', $product->id)
            ->get()
            ->keyBy('id');

        $keep = [];

        foreach ($variants as $i => $v) {
            $id = isset($v['id']) ? (int) $v['id'] : null;

            if ($id !== null && ! $existing->has($id)) {
                throw ValidationException::withMessages(["variants.$i.id" => ['Unknown variant.']]);
            }

            $variant = $id !== null ? $existing->get($id) : new ProductVariant();
            $price = $v['unit_price'] ?? null;

            $variant->forceFill([
                'product_id' => $product->id,
                'sku' => trim((string) $v['sku']),
                'name' => trim((string) $v['name']),
                'unit_price' => ($price === null || $price === '') ? null : $this->money($price),
                'is_active' => (bool) ($v['is_active'] ?? true),
                'sort_order' => $i,
            ])->save();

            $keep[] = $variant->id;
        }

        ProductVariant::query()
            ->where('product_id', $product->id)
            ->whereNotIn('id', $keep)
            ->delete();
    }




    /** Friendly pre-check (the unique indexes are the real guard): duplicates in the payload and in the tenant. */
    private function assertSkusFree(int $tenantId, ?int $productId, string $sku, array $variants): void
    {
        $errors = [];
        $productSku = mb_strtolower($sku);
        $variantSkus = array_map(fn (array $v) => mb_strtolower(trim((string) $v['sku'])), $variants);
        $candidates = array_values(array_unique(array_merge([$productSku], $variantSkus)));

        $takenByProducts = DB::table('products')
            ->where('tenant_id', $tenantId)
            ->whereNull('deleted_at')
            ->when($productId !== null, fn ($q) => $q->where('id', '!=', $productId))
            ->whereIn(DB::raw('lower(sku)'), $candidates)
            ->selectRaw('lower(sku) as s')
            ->pluck('s')
            ->all();

        $takenByVariants = DB::table('product_variants')
            ->where('tenant_id', $tenantId)
            ->when($productId !== null, fn ($q) => $q->where('product_id', '!=', $productId))
            ->whereIn(DB::raw('lower(sku)'), $candidates)
            ->selectRaw('lower(sku) as s')
            ->pluck('s')
            ->all();

        $taken = array_merge($takenByProducts, $takenByVariants);

        if (in_array($productSku, $taken, true)) {
            $errors['sku'] = ['This SKU is already in use.'];
        }

        $seen = [$productSku => true];
        foreach ($variantSkus as $i => $vsku) {
            if (isset($seen[$vsku])) {
                $errors["variants.$i.sku"] = ['Duplicate SKU in this product.'];
            } elseif (in_array($vsku, $taken, true)) {
                $errors["variants.$i.sku"] = ['This SKU is already in use.'];
            }
            $seen[$vsku] = true;
        }

        if ($errors !== []) {
            throw ValidationException::withMessages($errors);
        }
    }

    private function validMoney(mixed $value): bool
    {
        return is_numeric($value) && (float) $value >= 0 && (float) $value <= 9999999999.99;
    }

    private function money(mixed $value): string
    {
        return number_format((float) $value, 2, '.', '');
    }
}

<?php

namespace App\Http\Controllers\Api\Products;

use App\Domain\Products\Actions\DeleteProductAction;
use App\Domain\Products\Actions\SaveProductAction;
use App\Http\Controllers\Controller;
use App\Http\Requests\Products\ListProductsRequest;
use App\Http\Requests\Products\SaveProductRequest;
use App\Http\Resources\ProductResource;
use App\Models\Product;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\AnonymousResourceCollection;
use Illuminate\Http\Response;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Gate;


final class ProductController extends Controller
{
    public function index(ListProductsRequest $request): AnonymousResourceCollection
    {
        $v = $request->validated();
        $tenantId = (int) $request->user()->tenant_id;
        $q = trim((string) ($v['q'] ?? ''));

        $products = Product::query()
            ->where('tenant_id', $tenantId)
            ->withCount('variants')
            ->when($q !== '', function ($query) use ($q, $tenantId) {
                $like = '%'.addcslashes($q, '\\%_').'%';
                $query->where(function ($w) use ($like, $tenantId) {
                    $w->where('sku', 'ilike', $like)
                        ->orWhere('name', 'ilike', $like)
                        ->orWhereIn('id', DB::table('product_variants')
                            ->where('tenant_id', $tenantId)
                            ->where('sku', 'ilike', $like)
                            ->select('product_id'));
                });
            })
            ->when(($v['status'] ?? null) === 'active', fn ($query) => $query->where('is_active', true))
            ->when(($v['status'] ?? null) === 'inactive', fn ($query) => $query->where('is_active', false))
            ->when($v['tax_category'] ?? null, fn ($query, $category) => $query->where('tax_category', $category))
            ->orderBy($v['sort'] ?? 'name', $v['direction'] ?? 'asc')
            ->orderBy('id', $v['direction'] ?? 'asc') // stable tiebreaker so pages never overlap
            ->paginate($v['per_page'] ?? 25)
            ->withQueryString();

        return ProductResource::collection($products);
    }

    public function show(int $product): ProductResource
    {
        Gate::authorize('products.view');

        return new ProductResource(
            Product::query()
                ->with('variants')
                ->findOrFail($product),
        );
    }

    public function store(SaveProductRequest $request, SaveProductAction $action): JsonResponse
    {
        $product = $action->execute($request->user(), null, $request->validated());

        return (new ProductResource($product))->response()->setStatusCode(201);
    }

    public function update(SaveProductRequest $request, int $product, SaveProductAction $action): ProductResource
    {
        return new ProductResource($action->execute($request->user(), $product, $request->validated()));
    }

    public function destroy(Request $request, int $product, DeleteProductAction $action): Response
    {
        $action->execute($request->user(), $product);

        return response()->noContent();
    }
}

     * never name a class. Deliberately a non-enforcing morph map: enforceMorphMap()
     * would also break Spatie roles and Sanctum tokens, which morph to User.

     * @var array<string, class-string<Model>>

    public const TARGETS = [

        'contact' => Contact::class,
        'account' => Account::class,
        'deal' => Deal::class,
    ];

        'activityable_type',
        'activityable_id',
        'type',
        'subject',
        'body',
        'due_at',
        'owner_id',
    ];

    /** @return array<string, string> */

    {
        return [
            'activityable_id' => 'integer',
            'due_at' => 'datetime',
            'completed_at' => 'datetime',
            'overdue_at' => 'datetime',
        ];
    }

    public static function registerMorphMap(): void
    {
        Relation::morphMap(self::TARGETS);
    }

    /** @return class-string<Model>|null */
    public static function targetClass(string $alias): ?string
    {
        return self::TARGETS[$alias] ?? null;
    }

    /** The permission needed to see a target record (leads.view, contacts.view, ...). */
    public static function targetViewPermission(string $alias): string
    {
        return $alias . 's.view';
    }

    public static function ownerBelongsToTenant(int $userId): bool
    {
        if (! TenantContext::hasTenant()) {
            return false;
        }

        return User::query()
            ->whereKey($userId)
            ->where('tenant_id', TenantContext::id())
            ->exists();
    }

    public function activityable(): MorphTo
    {
        return $this->morphTo();
    }

    public function owner(): BelongsTo
    {
        return $this->belongsTo(User::class, 'owner_id');
    }
}

<?php

namespace App\Notifications;

use App\Models\Activity;
use Illuminate\Notifications\Messages\BroadcastMessage;
use Illuminate\Notifications\Notification;

/**
 * Sent to an activity's owner when activities:flag-overdue flags it (one send per
 * flag transition). Database + broadcast. The payload always carries tenant_id so
 * every read path can filter by tenant even if a user later belongs to several.
 */
class ActivityOverdueNotification extends Notification
{
    public const KIND = 'activity_overdue';

    public function __construct(private readonly Activity $activity)
    {
    }

    /** @return list<string> */
    public function via(object $notifiable): array
    {
        return ['database', 'broadcast'];
    }

    /** @return array<string, mixed> */
    public function toArray(object $notifiable): array
    {
        $alias = $this->activity->activityable_type;

        return [
            'kind' => self::KIND,
            'tenant_id' => (int) $this->activity->tenant_id,
            'activity_id' => (int) $this->activity->id,
            'activity_type' => $this->activity->type,
            'subject' => $this->activity->subject,
            'due_at' => $this->activity->due_at?->toIso8601String(),
            'target_type' => is_string($alias) && array_key_exists($alias, Activity::TARGETS) ? $alias : null,
            'target_id' => $this->activity->activityable_id !== null ? (int) $this->activity->activityable_id : null,
        ];
    }

    /**
     * Tiny payload; broadcast synchronously so the bell updates without needing a
     * queue worker. The client only uses the event as an "invalidate" signal.
     */
    public function toBroadcast(object $notifiable): BroadcastMessage
    {
        return (new BroadcastMessage($this->toArray($notifiable)))->onConnection('sync');
    }
}

function prodapiUser(Tenant $tenant, ?string $role = 'manager'): User
{
    prodapiSwitch($tenant);
    RoleCatalog::provision($tenant);

    $user = User::factory()->create(['tenant_id' => $tenant->id]);
    if ($role !== null) {
        $user->assignRole($role);
    }
    app(PermissionRegistrar::class)->forgetCachedPermissions();

it('prodapi shows and updates a product, replacing its variants', function () {
    $tenant = Tenant::factory()->create();
    $manager = prodapiUser($tenant);

    $keep = $product->variants[0]->id;
    Sanctum::actingAs($manager);

    $this->getJson("/api/v1/products/{$product->id}")
